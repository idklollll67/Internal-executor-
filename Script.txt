local function ShowNotification(message, accentColor)
    task.spawn(function()
        local notif = Instance.new("Frame")
        notif.Size = UDim2.new(1, 0, 0, 42)
        notif.BackgroundColor3 = Color3.fromRGB(16, 16, 24)
        notif.BackgroundTransparency = 1
        notif.Parent = NotifContainer

        corner.CornerRadius = UDim.new(0, 12)
        corner.Parent = notif

        local stroke = Instance.new("UIStroke")
        stroke.Color = accentColor or Color3.fromRGB(0, 240, 255)
        stroke.Transparency = 1
        stroke.Parent = notif

        local indicator = Instance.new("Frame")
        indicator.Size = UDim2.new(0, 4, 1, -16)
        indicator.Position = UDim2.new(0, 8, 0, 8)
        indicator.BackgroundColor3 = accentColor or Color3.fromRGB(0, 240, 255)
        indicator.BorderSizePixel = 0
        indicator.Parent = notif

        local indCorner = Instance.new("UICorner")
        indCorner.CornerRadius = UDim.new(1, 0)
        indCorner.Parent = indicator

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, -24, 1, 0)
        label.Position = UDim2.new(0, 20, 0, 0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamMedium
        label.Text = message
        label.TextColor3 = Color3.fromRGB(245, 245, 255)
        label.TextSize = 11
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = notif

        local ti = TweenInfo.new(0.4, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)
        TweenService:Create(notif, ti, {BackgroundTransparency = 0.1}):Play()
        TweenService:Create(stroke, ti, {Transparency = 0.4}):Play()

        task.wait(2.8)

        local fadeOut = TweenService:Create(notif, ti, {BackgroundTransparency = 1})
        TweenService:Create(stroke, ti, {Transparency = 1}):Play()
        TweenService:Create(label, ti, {TextTransparency = 1}):Play()
        TweenService:Create(indicator, ti, {BackgroundTransparency = 1}):Play()
        fadeOut:Play()
        fadeOut.Completed:Connect(function() notif:Destroy() end)
    end)
end

-- Draggable Toggle Logic
local dragging, dragInput, dragStart, startPos
OpenCloseBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = OpenCloseBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
OpenCloseBtn.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        OpenCloseBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- CONFIG SYSTEM HANDLING
local UserFileName = ConfigFolderName .. "/" .. LocalPlayer.Name .. "_SavedScripts.json"
local GlobalForcedFile = ConfigFolderName .. "/ForcedLockedScripts.json"

local ForcedLockedList = {}
pcall(function()
    if isfile(GlobalForcedFile) then
        local decoded = HttpService:JSONDecode(readfile(GlobalForcedFile))
        if type(decoded) == "table" then ForcedLockedList = decoded end
    end
end)

local SavedScriptsList = {
    {name = "Infinite Yield", code = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))()', locked = false}
}

for _, forcedItem in ipairs(ForcedLockedList) do
    local exists = false
    for _, item in ipairs(SavedScriptsList) do
        if item.name == forcedItem.name then exists = true; item.locked = true end
    end
    if not exists then table.insert(SavedScriptsList, {name = forcedItem.name, code = forcedItem.code, locked = true}) end
end

local function SaveConfigsToFile()
    pcall(function() writefile(UserFileName, HttpService:JSONEncode(SavedScriptsList)) end)
end

pcall(function()
    if isfile(UserFileName) then
        local decoded = HttpService:JSONDecode(readfile(UserFileName))
        if type(decoded) == "table" then SavedScriptsList = decoded end
    end
end)

            local req = request or syn.request or http_request

                Url = "https://scriptblox.com/api/script/search?q=" .. HttpService:UrlEncode(SearchBox.Text),

CreateMiscButton("Server Hop", function()
    ShowNotification("Scanning open game servers...", Color3.fromRGB(255, 220, 100))
    task.spawn(function()
        local success, servers = pcall(function()
            local req = request or syn.request or http_request
            if not req then return end
            local res = req({
                Url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100",
                Method = "GET"
            })
            if res and res.Body then return HttpService:JSONDecode(res.Body) end
        end)
        if success and servers and servers.data then
            for _, s in ipairs(servers.data) do
                if type(s) == "table" and s.id ~= game.JobId and s.playing < s.maxPlayers then
                    TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                    break
                end
            end
        end
    end)
end)

CreateMiscButton("Small Server Finder", function()
    ShowNotification("Locating low population instance...", Color3.fromRGB(255, 220, 100))
    task.spawn(function()
        local success, servers = pcall(function()
            local req = request or syn.request or http_request
            if not req then return end
            local res = req({
                Url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100",
                Method = "GET"
            })
            if res and res.Body then return HttpService:JSONDecode(res.Body) end
        end)
        if success and servers and servers.data then
            local targetServer = nil
            local minPlayers = math.huge
            for _, s in ipairs(servers.data) do
                if type(s) == "table" and s.id ~= game.JobId and s.playing < s.maxPlayers and s.playing < minPlayers then
                    minPlayers = s.playing
                    targetServer = s
                end
            end
            if targetServer then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, targetServer.id, LocalPlayer)
            end
        end
    end)
end)

    btn.Font = Enum.Font.GothamMedium

    corner.Parent = btn

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(40, 40, 65)
    stroke.Parent = btn

    btn.MouseButton1Click:Connect(function()
        isAntiAfkEnabled = not isAntiAfkEnabled
        if isAntiAfkEnabled then
            btn.Text = name .. ": On"
            btn.TextColor3 = Color3.fromRGB(100, 255, 150)
            ShowNotification("Anti AFK system enabled", Color3.fromRGB(100, 255, 150))

            btn.Text = name .. ": Off"

        end
        callback(isAntiAfkEnabled, btn)
    end)
end
