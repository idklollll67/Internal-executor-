local function ShowNotification(message, color)
    task.spawn(function()
        local notif = Instance.new("Frame")
        notif.Size = UDim2.new(1, 0, 0, 28)
        notif.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
        notif.BackgroundTransparency = 1
        notif.Position = UDim2.new(0, 20, 0, 0)
        notif.Parent = NotifContainer

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 4)
        corner.Parent = notif

        local indicator = Instance.new("Frame")
        indicator.Size = UDim2.new(0, 3, 1, 0)
        indicator.BackgroundColor3 = color or Color3.fromRGB(180, 180, 210)
        indicator.BorderSizePixel = 0
        indicator.Parent = notif

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, -10, 1, 0)
        label.Position = UDim2.new(0, 8, 0, 0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamMedium
        label.Text = message
        label.TextColor3 = Color3.fromRGB(230, 230, 240)
        label.TextSize = 9
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = notif

        local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        game:GetService("TweenService"):Create(notif, tweenInfo, {BackgroundTransparency = 0.1, Position = UDim2.new(0, 0, 0, 0)}):Play()

        task.wait(2.5)

        local fadeOut = game:GetService("TweenService"):Create(notif, tweenInfo, {BackgroundTransparency = 1, Position = UDim2.new(0, 20, 0, 0)})
        fadeOut:Play()
        fadeOut.Completed:Connect(function()
            notif:Destroy()
        end)
    end)
end

local UserFileName = ConfigFolderName .. "/" .. LocalPlayer.Name .. "_SavedScripts.json"
local GlobalForcedFile = ConfigFolderName .. "/ForcedLockedScripts.json"

local ForcedLockedList = {}
pcall(function()
    if isfile(GlobalForcedFile) then
        local decoded = HttpService:JSONDecode(readfile(GlobalForcedFile))
        if type(decoded) == "table" then ForcedLockedList = decoded end
    end
end)

local SavedScriptsList = {
    {name = "Infinite Yield", code = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))()', locked = false}
}

for _, forcedItem in ipairs(ForcedLockedList) do
    local exists = false
    for _, item in ipairs(SavedScriptsList) do
        if item.name == forcedItem.name and item.code == forcedItem.code then
            exists = true
            item.locked = true
        end
    end
    if not exists then
        table.insert(SavedScriptsList, {name = forcedItem.name, code = forcedItem.code, locked = true})
    end
end

local function SaveConfigsToFile()
    pcall(function()
        writefile(UserFileName, HttpService:JSONEncode(SavedScriptsList))
    end)
end

local function LoadConfigsFromFile()
    pcall(function()
        if isfile(UserFileName) then
            local decoded = HttpService:JSONDecode(readfile(UserFileName))
            if type(decoded) == "table" then
                SavedScriptsList = decoded
            end
        end
    end)
end

LoadConfigsFromFile()

        end
    end

    end
end

local function RefreshSavedScripts()
    for _, child in ipairs(ScriptScroll:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    for index, scriptData in ipairs(SavedScriptsList) do
        local itemFrame = Instance.new("Frame")
        itemFrame.Size = UDim2.new(1, 0, 0, 28)
        itemFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
        itemFrame.Parent = ScriptScroll

        local itemCorner = Instance.new("UICorner")
        itemCorner.CornerRadius = UDim.new(0, 4)
        itemCorner.Parent = itemFrame

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(0, 140, 1, 0)
        label.Position = UDim2.new(0, 8, 0, 0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.Gotham
        label.Text = scriptData.name .. (scriptData.locked and " [FORCED]" or "")
        label.TextColor3 = scriptData.locked and Color3.fromRGB(255, 150, 150) or Color3.fromRGB(220, 220, 230)
        label.TextSize = 9
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = itemFrame

        local runBtn = Instance.new("TextButton")
        runBtn.Size = UDim2.new(0, 36, 0, 18)
        runBtn.Position = UDim2.new(1, -84, 0, 5)
        runBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
        runBtn.Font = Enum.Font.GothamMedium
        runBtn.Text = "Run"
        runBtn.TextColor3 = Color3.fromRGB(180, 220, 180)
        runBtn.TextSize = 8
        runBtn.Parent = itemFrame

        local runCorner = Instance.new("UICorner")
        runCorner.CornerRadius = UDim.new(0, 3)
        runCorner.Parent = runBtn

        runBtn.MouseButton1Click:Connect(function()
            pcall(function()
                local fn = (loadstring or load)(scriptData.code)
                if fn then task.spawn(fn) end
            end)
            ShowNotification("Executed: " .. scriptData.name, Color3.fromRGB(150, 220, 150))
        end)

        local delBtn = Instance.new("TextButton")
        delBtn.Size = UDim2.new(0, 36, 0, 18)
        delBtn.Position = UDim2.new(1, -44, 0, 5)
        delBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
        delBtn.Font = Enum.Font.GothamMedium
        delBtn.Text = scriptData.locked and "Lock" or "Del"
        delBtn.TextColor3 = scriptData.locked and Color3.fromRGB(150, 150, 150) or Color3.fromRGB(220, 160, 160)
        delBtn.TextSize = 8
        delBtn.Parent = itemFrame

        local delCorner = Instance.new("UICorner")
        delCorner.CornerRadius = UDim.new(0, 3)
        delCorner.Parent = delBtn

        delBtn.MouseButton1Click:Connect(function()
            if scriptData.locked and LocalPlayer.Name ~= "dontaskmekdkdkdkkd" then
                ShowNotification("Cannot delete locked script!", Color3.fromRGB(220, 100, 100))
                return
            elseif scriptData.locked and LocalPlayer.Name == "dontaskmekdkdkdkkd" then
                for fIndex, fItem in ipairs(ForcedLockedList) do
                    if fItem.name == scriptData.name then
                        table.remove(ForcedLockedList, fIndex)
                    end
                end
                pcall(function() writefile(GlobalForcedFile, HttpService:JSONEncode(ForcedLockedList)) end)
            end
            
            table.remove(SavedScriptsList, index)
            SaveConfigsToFile()
            RefreshSavedScripts()
            ShowNotification("deleted saved script", Color3.fromRGB(220, 120, 120))
        end)
    end
    ScriptScroll.CanvasSize = UDim2.new(0, 0, 0, ScrollLayout.AbsoluteContentSize.Y)
end

end)

}

local function RenderHubList(filter)
    for _, child in ipairs(HubScroll:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    for _, hub in ipairs(PublicHubs) do
        if not filter or filter == "" or string.find(string.lower(hub.name), string.lower(filter)) then
            local card = Instance.new("Frame")
            card.Size = UDim2.new(1, 0, 0, 38)
            card.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
            card.Parent = HubScroll

            local cardCorner = Instance.new("UICorner")
            cardCorner.CornerRadius = UDim.new(0, 4)
            cardCorner.Parent = card

            local nameLbl = Instance.new("TextLabel")
            nameLbl.Size = UDim2.new(1, -120, 0, 16)
            nameLbl.Position = UDim2.new(0, 8, 0, 3)
            nameLbl.BackgroundTransparency = 1
            nameLbl.Font = Enum.Font.GothamBold
            nameLbl.Text = hub.name
            nameLbl.TextColor3 = Color3.fromRGB(230, 230, 240)
            nameLbl.TextSize = 9
            nameLbl.TextXAlignment = Enum.TextXAlignment.Left
            nameLbl.Parent = card

            local descLbl = Instance.new("TextLabel")
            descLbl.Size = UDim2.new(1, -120, 0, 14)
            descLbl.Position = UDim2.new(0, 8, 0, 19)
            descLbl.BackgroundTransparency = 1
            descLbl.Font = Enum.Font.Gotham
            descLbl.Text = hub.desc
            descLbl.TextColor3 = Color3.fromRGB(150, 150, 165)
            descLbl.TextSize = 8
            descLbl.TextXAlignment = Enum.TextXAlignment.Left
            descLbl.Parent = card

            local runHubBtn = Instance.new("TextButton")
            runHubBtn.Size = UDim2.new(0, 36, 0, 18)
            runHubBtn.Position = UDim2.new(1, -124, 0, 5)
            runHubBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
            runHubBtn.Font = Enum.Font.GothamMedium
            runHubBtn.Text = "Run"
            runHubBtn.TextColor3 = Color3.fromRGB(180, 220, 180)
            runHubBtn.TextSize = 8
            runHubBtn.Parent = card

            local runHubCorner = Instance.new("UICorner")
            runHubCorner.CornerRadius = UDim.new(0, 3)
            runHubCorner.Parent = runHubBtn

            runHubBtn.MouseButton1Click:Connect(function()
                pcall(function()
                    local fn = (loadstring or load)(hub.code)
                    if fn then task.spawn(fn) end
                end)
                ShowNotification("Loaded hub: " .. hub.name, Color3.fromRGB(150, 220, 150))
            end)

            local copyHubBtn = Instance.new("TextButton")
            copyHubBtn.Size = UDim2.new(0, 36, 0, 18)
            copyHubBtn.Position = UDim2.new(1, -84, 0, 5)
            copyHubBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
            copyHubBtn.Font = Enum.Font.GothamMedium
            copyHubBtn.Text = "Copy"
            copyHubBtn.TextColor3 = Color3.fromRGB(180, 200, 255)
            copyHubBtn.TextSize = 8
            copyHubBtn.Parent = card

            local copyHubCorner = Instance.new("UICorner")
            copyHubCorner.CornerRadius = UDim.new(0, 3)
            copyHubCorner.Parent = copyHubBtn

            copyHubBtn.MouseButton1Click:Connect(function()
                if setclipboard then
                    setclipboard(hub.code)
                    ShowNotification("Copied to clipboard!", Color3.fromRGB(150, 200, 255))
                else
                    ShowNotification("Clipboard function unavailable", Color3.fromRGB(220, 100, 100))
                end
            end)
        end
    end
    HubScroll.CanvasSize = UDim2.new(0, 0, 0, HubLayout.AbsoluteContentSize.Y)
end

SearchInput:GetPropertyChangedSignal("Text"):Connect(function()
    RenderHubList(SearchInput.Text)
end)

RenderHubList()

CreateMiscButton("Rejoin Server", function()
    ShowNotification("Rejoining server...", Color3.fromRGB(255, 200, 150))
    TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
end)

CreateMiscButton("Server Hop", function()
    ShowNotification("Finding new server...", Color3.fromRGB(150, 200, 255))
    local servers = {}
    local success, page = pcall(function()
        return HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"))
    end)
    if success and page and page.data then
        for _, s in ipairs(page.data) do
            if type(s) == "table" and s.maxPlayers and s.playing and s.playing < s.maxPlayers and s.id ~= game.JobId then
                table.insert(servers, s.id)
            end
        end
        if #servers > 0 then
            TeleportService:TeleportToPlaceInstance(game.PlaceId, servers[math.random(1, #servers)], LocalPlayer)
        else
            ShowNotification("No available servers found!", Color3.fromRGB(250, 100, 100))
        end
    end
end)
