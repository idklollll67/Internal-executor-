    task.spawn(function()
        local notif = Instance.new("Frame")
        notif.Size = UDim2.new(1, 0, 0, 36)
        notif.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
        notif.BackgroundTransparency = 1
        notif.Parent = NotifContainer

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 8)
        corner.Parent = notif

        local stroke = Instance.new("UIStroke")
        stroke.Color = Color3.fromRGB(45, 45, 60)
        stroke.Transparency = 1
        stroke.Parent = notif

        local indicator = Instance.new("Frame")
        indicator.Size = UDim2.new(0, 4, 1, -12)
        indicator.Position = UDim2.new(0, 6, 0, 6)
        indicator.BackgroundColor3 = color or Color3.fromRGB(130, 110, 255)
        indicator.BorderSizePixel = 0
        indicator.Parent = notif

        local indCorner = Instance.new("UICorner")
        indCorner.CornerRadius = UDim.new(1, 0)
        indCorner.Parent = indicator

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, -20, 1, 0)
        label.Position = UDim2.new(0, 18, 0, 0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamMedium
        label.Text = message
        label.TextColor3 = Color3.fromRGB(235, 235, 245)
        label.TextSize = 10
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = notif

        local ts = game:GetService("TweenService")
        local ti = TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
        ts:Create(notif, ti, {BackgroundTransparency = 0.05}):Play()
        ts:Create(stroke, ti, {Transparency = 0.4}):Play()

        task.wait(2.5)

        local fadeOut = ts:Create(notif, ti, {BackgroundTransparency = 1})
        ts:Create(stroke, ti, {Transparency = 1}):Play()
        ts:Create(label, ti, {TextTransparency = 1}):Play()
        fadeOut:Play()
        fadeOut.Completed:Connect(function()
            notif:Destroy()
        end)
    end)
end

local dragging, dragInput, dragStart, startPos
OpenCloseBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = OpenCloseBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

OpenCloseBtn.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        OpenCloseBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local UserFileName = ConfigFolderName .. "/" .. LocalPlayer.Name .. "_SavedScripts.json"
local GlobalForcedFile = ConfigFolderName .. "/ForcedLockedScripts.json"

local ForcedLockedList = {}
pcall(function()
    if isfile(GlobalForcedFile) then
        local decoded = HttpService:JSONDecode(readfile(GlobalForcedFile))
        if type(decoded) == "table" then ForcedLockedList = decoded end
    end
end)

local SavedScriptsList = {
    {name = "Infinite Yield", code = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))()', locked = false}
}

for _, forcedItem in ipairs(ForcedLockedList) do
    local exists = false
    for _, item in ipairs(SavedScriptsList) do
        if item.name == forcedItem.name and item.code == forcedItem.code then
            exists = true
            item.locked = true
        end
    end
    if not exists then
        table.insert(SavedScriptsList, {name = forcedItem.name, code = forcedItem.code, locked = true})
    end
end

local function SaveConfigsToFile()
    pcall(function()
        writefile(UserFileName, HttpService:JSONEncode(SavedScriptsList))
    end)
end

local function LoadConfigsFromFile()
    pcall(function()
        if isfile(UserFileName) then
            local decoded = HttpService:JSONDecode(readfile(UserFileName))
            if type(decoded) == "table" then
                SavedScriptsList = decoded
            end
        end
    end)
end

LoadConfigsFromFile()

for _, forcedItem in ipairs(ForcedLockedList) do
    local found = false
    for _, item in ipairs(SavedScriptsList) do
        if item.name == forcedItem.name then
            found = true
            item.locked = true
        end
    end
    if not found then
        table.insert(SavedScriptsList, {name = forcedItem.name, code = forcedItem.code, locked = true})
    end
end
