-- CHAT COMMANDS FUNCTIONALITY (!tpto, !reset, !lock, !unlock, !fix)
LocalPlayer.Chatted:Connect(function(message)
    local args = {}
    for word in message:gmatch("%S+") do
        table.insert(args, word)
    end
    
    local cmd = args[1] and args[1]:lower() or ""
    
    if cmd == "!tpto" then
        local targetName = args[2]
        if not targetName then
            ShowNotification("Usage: !tpto <player>", Color3.fromRGB(240, 140, 140))
            return
        end
        
        local foundPlayer = nil
        targetName = targetName:lower()
        
        for _, p in ipairs(Players:GetPlayers()) do
            if p.Name:lower() == targetName or (p.DisplayName and p.DisplayName:lower() == targetName) then
                foundPlayer = p
                break
            end
        end
        
        if not foundPlayer then
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Name:lower():find(targetName, 1, true) or (p.DisplayName and p.DisplayName:lower():find(targetName, 1, true)) then
                    foundPlayer = p
                    break
                end
            end
        end
        
        if foundPlayer and foundPlayer.Character and foundPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if myRoot then
                myRoot.CFrame = foundPlayer.Character.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
                ShowNotification("Teleported to " .. foundPlayer.Name, Color3.fromRGB(150, 230, 150))
            end
        else
            ShowNotification("Player not found or has no character!", Color3.fromRGB(240, 140, 140))
        end
    elseif cmd == "!reset" then
        local char = LocalPlayer.Character
        if char and char:FindFirstChildOfClass("Humanoid") then
            char:FindFirstChildOfClass("Humanoid").Health = 0
            ShowNotification("Reset character via chat", Color3.fromRGB(230, 180, 140))
        end
    elseif cmd == "!lock" then
        isUiLocked = true
        ShowNotification("UI Locked via chat command", Color3.fromRGB(240, 180, 140))
    elseif cmd == "!unlock" then
        isUiLocked = false
        ShowNotification("UI Unlocked via chat command", Color3.fromRGB(150, 230, 150))
    elseif cmd == "!fix" then
        isUiLocked = false
        ScreenGui.Parent = (gethui and gethui()) or LocalPlayer:WaitForChild("PlayerGui")
        MainFrame.Position = UDim2.new(0.5, -280, 0.5, -170)
        OpenCloseBtn.Position = UDim2.new(0, 18, 0.5, -22)
        MainFrame.Visible = true
        ShowNotification("UI Force Reset to Center Screen", Color3.fromRGB(150, 230, 150))
    end
end)

local UserFileName = ConfigFolderName .. "/" .. LocalPlayer.Name .. "_SavedScripts.json"
local GlobalForcedFile = ConfigFolderName .. "/ForcedLockedScripts.json"

local ForcedLockedList = {}
pcall(function()
    if isfile(GlobalForcedFile) then
        local decoded = HttpService:JSONDecode(readfile(GlobalForcedFile))
        if type(decoded) == "table" then ForcedLockedList = decoded end
    end
end)

local SavedScriptsList = {
    {name = "Infinite Yield", code = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source"))()', locked = false}
}

for _, forcedItem in ipairs(ForcedLockedList) do
    local exists = false
    for _, item in ipairs(SavedScriptsList) do
        if item.name == forcedItem.name and item.code == forcedItem.code then
            exists = true
            item.locked = true
        end
    end
    if not exists then
        table.insert(SavedScriptsList, {name = forcedItem.name, code = forcedItem.code, locked = true})
    end
end

local function SaveConfigsToFile()
    pcall(function()
        writefile(UserFileName, HttpService:JSONEncode(SavedScriptsList))
    end)
end

local function LoadConfigsFromFile()
    pcall(function()
        if isfile(UserFileName) then
            local decoded = HttpService:JSONDecode(readfile(UserFileName))
            if type(decoded) == "table" then
                SavedScriptsList = decoded
            end
        end
    end)
end

        local Terrain = workspace:FindFirstChildOfClass("Terrain")
        
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 9e9
        Lighting.Brightness = 1
        Lighting.ClockTime = 14
        for _, v in ipairs(Lighting:GetChildren()) do
            pcall(function() v:Destroy() end)
        end

        if Terrain then
            Terrain.WaterWaveSize = 0
            Terrain.WaterWaveSpeed = 0
            Terrain.WaterReflectance = 0
            Terrain.WaterTransparency = 1
            pcall(function() Terrain.Decoration = false end)
        end

        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
                v.CastShadow = false
                v.Color = Color3.fromRGB(130, 130, 130)
            elseif v:IsA("Decal") or v:IsA("Texture") or v:IsA("SpecialMesh") or v:IsA("MeshPart") then
                pcall(function() v:Destroy() end)
            elseif v:IsA("ParticleEmitter") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") or v:IsA("Trail") or v:IsA("Beam") or v:IsA("Explosion") then
                pcall(function() v:Destroy() end)
            elseif v:IsA("PostEffect") or v:IsA("Atmosphere") or v:IsA("Sky") or v:IsA("Clouds") then
                pcall(function() v:Destroy() end)
            end
        end

        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        pcall(function()
            UserSettings():GetService("UserGameSettings").SavedQualityLevel = Enum.SavedQualityLevel.Level1
            UserSettings():GetService("UserGameSettings").MasterVolume = 0.3
            UserSettings():GetService("UserGameSettings").Fullscreen = true
        end)
    end)
    ShowNotification("Maximum FPS Boost Active!", Color3.fromRGB(150, 230, 150))
end)
